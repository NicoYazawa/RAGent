# 项目国际化（中英双语）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 前端 UI 支持中/英切换（vue-i18n），后端用户可见文本（SSE 决策链、API 错误、LLM 提示词）跟随请求语言，全流程中英双语一致。

**Architecture:** 前端 vue-i18n 双语言目录 + pinia locale store（localStorage 持久化、默认跟浏览器语言）；所有 fetch 经共享 `apiJson` 自动带 `Accept-Language` 头；后端 `app/i18n.py` 提供 `resolve_locale/t` + 中间件注入 ContextVar，orchestrator 的 SSE 文本与 routers 错误消息走目录取文案；LLM 提示词按**提问语言**选择 zh/en 两套（答案语言跟随提问，与 UI 语言解耦）。

**Tech Stack:** vue-i18n ^11 / pinia / vitest（新增前端测试）/ FastAPI ContextVar / pytest

**Spec:** 本文档（源自 2026-10-19 与用户确认的全量范围：A 前端 + C 后端语言跟随 + D 测试；B 注释英文化为独立后续计划）。翻译资产已就绪：`docs/i18n/glossary.tsv`（71 对术语）、`frontend/src/i18n/candidates/{strings.json,translations.json,review.csv}`（280 条候选与译文）。

## Global Constraints

- 默认体验不变：中文浏览器 + 无 localStorage → zh-CN，全部现有行为逐字节保持
- 语言目录 key 树必须严格一致（Task 11 有 parity 测试把关）；缺失 key 时回退 zh-CN 并 console.warn
- 占位符 `{0}`/`{n}`/`%s`/`[n]` 在翻译与代码替换中不得丢失或改名
- 术语以 `docs/i18n/glossary.tsv` 为准（切片=Chunk、知识库=Knowledge Base、检索路=Retrieval Lane 等）
- LLM 答案语言跟随提问语言（CJK 占比 ≥30% 判中文），与 UI 语言无关——README 需写明
- DB 存量决策日志的 detail 为渲染后文本，历史记录保持原语言（不做迁移，文档说明）
- 项目非 git 仓库，各 Task 收尾用「全量测试 + 构建」代替 commit
- 后端 105 个既有测试必须始终全绿；新增测试随 Task 落地

- [ ] **Step 3: 创建 `frontend/src/stores/locale.ts`**

```ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import i18n, { DEFAULT_LOCALE, SUPPORTED_LOCALES, type Locale } from '@/i18n'

const STORAGE_KEY = 'ragent-locale'

function normalize(v: string | null): Locale {
  if (v && (SUPPORTED_LOCALES as readonly string[]).includes(v)) return v as Locale
  // 浏览器语言前缀匹配（en-US → en，zh-CN → zh）
  if (v) {
    const lower = v.toLowerCase()
    if (lower.startsWith('zh')) return 'zh-CN'
    if (lower.startsWith('en')) return 'en-US'
  }
  return DEFAULT_LOCALE
}

export const useLocaleStore = defineStore('locale', () => {
  const locale = ref<Locale>(DEFAULT_LOCALE)

  function apply(v: Locale) {
    locale.value = v
    i18n.global.locale.value = v
    document.documentElement.lang = v === 'zh-CN' ? 'zh-CN' : 'en'
  }

  function initLocale() {
    const stored = typeof localStorage !== 'undefined' ? localStorage.getItem(STORAGE_KEY) : null
    const nav = typeof navigator !== 'undefined' ? navigator.language : null
    apply(normalize(stored ?? nav))
  }

  function setLocale(v: Locale) {
    apply(v)
    if (typeof localStorage !== 'undefined') localStorage.setItem(STORAGE_KEY, v)
  }

  return { locale, initLocale, setLocale }
})
```

```python
"""后端国际化 — 语言协商 / 消息目录 / ContextVar 中间件（C 方案：Accept-Language）

SSE 决策链文本、API 错误消息、LLM 提示词的选择都经这里；
UI 静态文案在前端 vue-i18n，不经过本模块。
"""
import logging
import re
from contextvars import ContextVar
from typing import Literal

from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
from starlette.responses import Response

logger = logging.getLogger("agentic.i18n")

Locale = Literal["zh-CN", "en-US"]
SUPPORTED = ("zh-CN", "en-US")
DEFAULT_LOCALE: Locale = "zh-CN"

# key -> {"zh-CN": ..., "en-US": ...}；Task 9/10 注入 sse/api/prompt 文案
MESSAGES: dict[str, dict[str, str]] = {}

_locale_ctx: ContextVar[str] = ContextVar("locale", default=DEFAULT_LOCALE)


def resolve_locale(accept_language: str | None) -> Locale:
    """解析 Accept-Language；不支持/缺失回落 zh-CN"""
    if not accept_language:
        return DEFAULT_LOCALE
    for part in accept_language.split(","):
        tag = part.split(";")[0].strip().lower()
        if not tag:
            continue
        if tag.startswith("zh"):
            return "zh-CN"
        if tag.startswith("en"):
            return "en-US"
    return DEFAULT_LOCALE


def current_locale() -> str:
    return _locale_ctx.get()


def set_locale(v: str) -> None:
    _locale_ctx.set(v if v in SUPPORTED else DEFAULT_LOCALE)


def t(key: str, locale: str | None = None, **params) -> str:
    """按语言取文案；缺 en 回落 zh；再缺返回 key 并告警"""
    loc = locale or current_locale()
    entry = MESSAGES.get(key)
    if entry is None:
        logger.warning("i18n 缺失 key：%s", key)
        return key
    text = entry.get(loc) or entry.get(DEFAULT_LOCALE)
    if text is None:
        logger.warning("i18n key %s 无任何语言文案", key)
        return key
    if params:
        try:
            text = text.format(**params)
        except (KeyError, IndexError, ValueError) as e:
            logger.warning("i18n 文案格式化失败 %s：%s", key, e)
    return text


class LocaleMiddleware(BaseHTTPMiddleware):
    """请求入口解析 Accept-Language → ContextVar（异步生成器内可见）"""

    async def dispatch(self, request: Request, call_next) -> Response:
        set_locale(resolve_locale(request.headers.get("accept-language")))
        return await call_next(request)
```

@dataclass(frozen=True)
class PromptBundle:
    ROUTER_SYS: str
    PLANNER_SYS: str
    GRADER_SYS: str
    CRAG_SYS: str
    REFLECT_SYS: str
    NL2SQL_SYS: str
    GENERATE_SYS: str
    CHAT_SYS: str
    router_user: object
    planner_user: object
    grader_user: object
    crag_user: object
    reflect_user: object
    nl2sql_user: object
    generate_user: object


def detect_lang(text: str) -> Lang:
    """CJK 字符占比 ≥30% 判中文；空文本回落默认中文"""
    if not text:
        return "zh-CN"
    cjk = sum(1 for ch in text if "\u4e00" <= ch <= "\u9fff")
    return "zh-CN" if cjk / max(len(text), 1) >= 0.3 else "en-US"


PROMPTS: dict[Lang, PromptBundle] = {
    "zh-CN": PromptBundle(ROUTER_SYS=_ZH_ROUTER_SYS, ..., generate_user=generate_user_zh),
    "en-US": PromptBundle(ROUTER_SYS=_EN_ROUTER_SYS, ..., generate_user=generate_user_en),
}


def prompts_for(text: str) -> PromptBundle:
    return PROMPTS[detect_lang(text)]

function keysOf(obj: unknown, prefix = ''): string[] {
  if (typeof obj !== 'object' || obj === null) return [prefix]
  return Object.entries(obj).flatMap(([k, v]) => keysOf(v, prefix ? `${prefix}.${k}` : k))
}

```python
"""端到端双语验证：同一问题分别以 Accept-Language: zh-CN / en-US 请求，比对 SSE 决策链语言"""
import json
import sys
import urllib.request

BASE = sys.argv[1] if len(sys.argv) > 1 else "http://localhost"


def ask(lang: str, question: str) -> dict:
    s = json.load(urllib.request.urlopen(urllib.request.Request(
        f"{BASE}/api/chat/sessions", method="POST", data=b"{}",
        headers={"Content-Type": "application/json", "Accept-Language": lang}), timeout=30))
    req = urllib.request.Request(
        f"{BASE}/api/chat/sessions/{s['id']}/ask", method="POST",
        data=json.dumps({"question": question}).encode(),
        headers={"Content-Type": "application/json", "Accept-Language": lang})
    details, answer, citations, grades = [], "", [], []
    etype = None
    for raw in urllib.request.urlopen(req, timeout=180):
        line = raw.decode("utf-8").rstrip("\n")
        if line.startswith("event: "):
            etype = line[7:]
        elif line.startswith("data: ") and etype:
            d = json.loads(line[6:])
            if etype == "retrieve" or etype == "grade" or etype == "router":
                details.append(d.get("detail", ""))
            elif etype == "generate":
                answer += d.get("delta", "")
            elif etype == "citation":
                citations = d.get("citations", [])
            elif etype == "grade":
                grades.append((d.get("score"), d.get("verdict")))
            etype = None
    return {"details": details, "answer": answer, "citations": citations, "grades": grades}


zh = ask("zh-CN", "迟到是怎么认定和处罚的？")
en = ask("en-US", "How is lateness handled?")
mixed = ask("en-US", "迟到是怎么认定和处罚的？")  # 英文 UI + 中文提问

def has_cjk(s: str) -> bool:
    return any("\u4e00" <= c <= "\u9fff" for c in s)

checks = [
    ("中文请求→中文决策链", has_cjk(" ".join(zh["details"]))),
    ("英文请求→英文决策链", not has_cjk(" ".join(en["details"]))),
    ("英文请求→英文答案", not has_cjk(en["answer"])),
    ("中文请求→中文答案", has_cjk(zh["answer"])),
    ("英文UI+中文提问→中文答案（语言跟随提问）", has_cjk(mixed["answer"])),
    ("双语引用均命中考勤制度", all(c["doc_name"] == "员工考勤与休假管理制度" for c in zh["citations"] + en["citations"])),
    ("双语评分均通过", all(g[1] == "pass" for g in zh["grades"] + en["grades"])),
]
ok = True
for name, passed in checks:
    print(f"  {'[OK]' if passed else '[FAIL]':<7} {name}")
    ok = ok and passed
print("RESULT:", "PASS" if ok else "FAIL")
sys.exit(0 if ok else 1)
```
